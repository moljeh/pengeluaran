const FOLDER_NAME = 'Nota Pengeluaran App'; 
const SHEET_NAME = 'Sheet1'; 

function getOrCreateFolder(folderName) {
  const folders = DriveApp.getFoldersByName(folderName);
  if (folders.hasNext()) { return folders.next(); } 
  else { return DriveApp.createFolder(folderName); }
}

function doGet(e) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
    let total = 0;
    let records = [];
    
    if (sheet && sheet.getLastRow() > 1) {
      const data = sheet.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        let jumlah = parseFloat(data[i][3]); 
        if (!isNaN(jumlah)) { total += jumlah; }
        records.push({
          row: i + 1,
          tanggal: data[i][1] || '-',
          kategori: data[i][2] || '-',
          jumlah: jumlah || 0,
          keterangan: data[i][4] || '-'
        });
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'success', total: total, data: records })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName(SHEET_NAME);
    
    if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); }
    if (sheet.getLastRow() === 0) {
      const headers = ["Waktu Timestamp", "Tanggal", "Kategori", "Jumlah (Rp)", "Keterangan", "Link Nota"];
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#10b981").setFontColor("white");
    }
    
    // 1. LOGIKA HAPUS DATA (SATUAN)
    if (data.action === 'delete') {
       sheet.deleteRow(data.row);
       return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Data dihapus!' })).setMimeType(ContentService.MimeType.JSON);
    }
    // 2. LOGIKA EDIT DATA
    else if (data.action === 'edit') {
       sheet.getRange(data.row, 2).setValue(data.tanggal);
       sheet.getRange(data.row, 3).setValue(data.kategori);
       sheet.getRange(data.row, 4).setValue(data.jumlah);
       sheet.getRange(data.row, 5).setValue(data.keterangan);
       return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Data diperbarui!' })).setMimeType(ContentService.MimeType.JSON);
    }
    // 3. LOGIKA RESET SEMUA DATA (BARU)
    else if (data.action === 'reset_all') {
       const lastRow = sheet.getLastRow();
       if (lastRow > 1) {
         // Menghapus mulai dari baris ke-2 hingga baris terakhir (menyisakan header)
         sheet.deleteRows(2, lastRow - 1);
       }
       return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Semua data berhasil dikosongkan!' })).setMimeType(ContentService.MimeType.JSON);
    }
    // 4. LOGIKA TAMBAH DATA BARU
    else {
      let fileUrl = '';
      if (data.gambarBase64 && data.namaFile) {
        const targetFolder = getOrCreateFolder(FOLDER_NAME);
        const contentType = data.gambarBase64.substring(5, data.gambarBase64.indexOf(';'));
        const bytes = Utilities.base64Decode(data.gambarBase64.split(',')[1]);
        const blob = Utilities.newBlob(bytes, contentType, data.namaFile);
        const file = targetFolder.createFile(blob);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        fileUrl = file.getUrl();
      }
      sheet.appendRow([new Date(), data.tanggal, data.kategori, data.jumlah, data.keterangan, fileUrl]);
      return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Data berhasil disimpan!' })).setMimeType(ContentService.MimeType.JSON);
    }
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
